cmake_minimum_required(VERSION 3.22)
project(usb_camera LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
# The selected-app recipe does not inherit pkgconfig; isolate target lookup.
if(CMAKE_CROSSCOMPILING AND CMAKE_SYSROOT)
    set(ENV{PKG_CONFIG_SYSROOT_DIR} "${CMAKE_SYSROOT}")
    set(ENV{PKG_CONFIG_LIBDIR} "${CMAKE_SYSROOT}/usr/lib/pkgconfig:${CMAKE_SYSROOT}/usr/share/pkgconfig")
    set(ENV{PKG_CONFIG_PATH} "")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-app-1.0 gstreamer-video-1.0)
add_executable(usb-camera src/main.cpp)
target_compile_features(usb-camera PRIVATE cxx_std_17)
target_compile_options(usb-camera PRIVATE -Wall -Wextra -Werror)
target_link_libraries(usb-camera PRIVATE PkgConfig::APP)
install(TARGETS usb-camera RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/usb-camera.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/usb-camera.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/usb-camera)
if(BUILD_TESTING)
    add_executable(camera-lifecycle tests/lifecycle.cpp)
    target_compile_features(camera-lifecycle PRIVATE cxx_std_17)
    target_compile_options(camera-lifecycle PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(camera-lifecycle PRIVATE PkgConfig::APP)
    add_test(NAME lifecycle COMMAND camera-lifecycle)
    add_test(NAME self-test COMMAND usb-camera --self-test)
    add_test(NAME invalid-duration COMMAND usb-camera --headless --duration nan)
    set_tests_properties(invalid-duration PROPERTIES WILL_FAIL TRUE)
    add_test(NAME unknown-option COMMAND usb-camera --unknown)
    set_tests_properties(unknown-option PROPERTIES WILL_FAIL TRUE)
endif()
