cmake_minimum_required(VERSION 3.22)
project(bird_classifier LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
# The generic app recipe inherits cmake, not pkgconfig.bbclass.
# Keep target .pc files separate from host libraries during cross builds.
if(CMAKE_CROSSCOMPILING AND CMAKE_SYSROOT)
    set(ENV{PKG_CONFIG_SYSROOT_DIR} "${CMAKE_SYSROOT}")
    set(ENV{PKG_CONFIG_LIBDIR} "${CMAKE_SYSROOT}/usr/lib/pkgconfig:${CMAKE_SYSROOT}/usr/share/pkgconfig")
    set(ENV{PKG_CONFIG_PATH} "")
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0)
find_path(QNN_INCLUDE_DIR QnnInterface.h PATH_SUFFIXES QNN REQUIRED)
add_executable(bird-classifier src/main.cpp src/qnn.cpp)
target_compile_features(bird-classifier PRIVATE cxx_std_17)
target_compile_options(bird-classifier PRIVATE -Wall -Wextra -Werror)
target_include_directories(bird-classifier PRIVATE ${QNN_INCLUDE_DIR})
target_link_libraries(bird-classifier PRIVATE PkgConfig::APP Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS bird-classifier RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/bird-classifier.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/bird-classifier.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/bird-classifier)
if(BUILD_TESTING)
    add_test(NAME self-test COMMAND bird-classifier --self-test)
    add_test(NAME invalid-repeat COMMAND bird-classifier --repeat -1)
    set_tests_properties(invalid-repeat PROPERTIES WILL_FAIL TRUE)
endif()
