cmake_minimum_required(VERSION 3.22)
project(imx708_camera LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(IMX708_WITH_QMMF "Build the Qualcomm camera-service backend" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UI REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 gstreamer-app-1.0)
add_library(camera_core src/core.cpp)
target_include_directories(camera_core PUBLIC src)
target_compile_features(camera_core PUBLIC cxx_std_17)
target_compile_options(camera_core PRIVATE -Wall -Wextra -Werror)
add_executable(imx708-camera src/main.cpp src/camera.cpp)
target_link_libraries(imx708-camera PRIVATE camera_core PkgConfig::UI)
target_compile_options(imx708-camera PRIVATE -Wall -Wextra -Werror)
if(IMX708_WITH_QMMF)
    pkg_check_modules(QMMF REQUIRED IMPORTED_TARGET qmmf_camera_metadata)
    find_library(QMMF_CLIENT qmmf_recorder_client REQUIRED)
    target_link_libraries(imx708-camera PRIVATE PkgConfig::QMMF ${QMMF_CLIENT})
    target_compile_definitions(imx708-camera PRIVATE IMX708_WITH_QMMF=1)
endif()
install(TARGETS imx708-camera RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/imx708-camera.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/imx708-camera)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/imx708-camera/docs)
if(BUILD_TESTING)
    add_executable(camera-core-test tests/core_test.cpp)
    target_link_libraries(camera-core-test PRIVATE camera_core)
    add_test(NAME camera-core COMMAND camera-core-test)
    add_test(NAME no-hardware-self-test COMMAND imx708-camera --self-test)
endif()
