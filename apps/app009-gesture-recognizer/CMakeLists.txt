cmake_minimum_required(VERSION 3.22)
project(gesture_recognizer LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_path(QNN_INCLUDE_DIR QnnInterface.h PATH_SUFFIXES QNN REQUIRED)
add_executable(npu-probe tests/npu_probe.cpp src/qnn.cpp)
target_compile_features(npu-probe PRIVATE cxx_std_17)
target_compile_options(npu-probe PRIVATE -Wall -Wextra -Werror)
target_include_directories(npu-probe PRIVATE src ${QNN_INCLUDE_DIR})
target_link_libraries(npu-probe PRIVATE ${CMAKE_DL_LIBS})
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-pbutils-1.0)
add_executable(gesture-recognizer src/main.cpp src/qnn.cpp)
target_compile_features(gesture-recognizer PRIVATE cxx_std_17)
target_compile_options(gesture-recognizer PRIVATE -Wall -Wextra -Werror)
target_include_directories(gesture-recognizer PRIVATE ${QNN_INCLUDE_DIR})
target_link_libraries(gesture-recognizer PRIVATE PkgConfig::APP Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS gesture-recognizer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/gesture-recognizer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/gesture-recognizer.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/gesture-recognizer)
if(BUILD_TESTING)
    add_executable(core-test tests/core_test.cpp)
    target_compile_features(core-test PRIVATE cxx_std_17)
    target_compile_options(core-test PRIVATE -Wall -Wextra -Werror)
    target_include_directories(core-test PRIVATE src)
    add_test(NAME core COMMAND core-test)
    add_test(NAME self-test COMMAND gesture-recognizer --self-test)
    add_test(NAME invalid-duration COMMAND gesture-recognizer --duration nan)
    set_tests_properties(invalid-duration PROPERTIES WILL_FAIL TRUE)
endif()
if(BUILD_TESTING)
    add_executable(contract-probe tests/contract_probe.cpp)
    target_compile_features(contract-probe PRIVATE cxx_std_17)
    target_include_directories(contract-probe PRIVATE src)
endif()
add_executable(qnn-batch tests/qnn_batch.cpp src/qnn.cpp)
target_compile_features(qnn-batch PRIVATE cxx_std_17)
target_compile_options(qnn-batch PRIVATE -Wall -Wextra -Werror)
target_include_directories(qnn-batch PRIVATE src ${QNN_INCLUDE_DIR})
target_link_libraries(qnn-batch PRIVATE ${CMAKE_DL_LIBS})
add_executable(qnn-multi-probe tests/qnn_multi_probe.cpp src/qnn.cpp)
target_compile_features(qnn-multi-probe PRIVATE cxx_std_17)
target_compile_options(qnn-multi-probe PRIVATE -Wall -Wextra -Werror)
target_include_directories(qnn-multi-probe PRIVATE src ${QNN_INCLUDE_DIR})
target_link_libraries(qnn-multi-probe PRIVATE ${CMAKE_DL_LIBS})
