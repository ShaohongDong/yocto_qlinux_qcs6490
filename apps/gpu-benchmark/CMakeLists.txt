cmake_minimum_required(VERSION 3.22)
project(gpu_benchmark LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
option(GPU_CORE_ONLY "Build hardware-independent tests only" OFF)
add_library(gpu_metrics src/core.cpp)
target_include_directories(gpu_metrics PUBLIC src)
target_compile_features(gpu_metrics PUBLIC cxx_std_17)
target_compile_options(gpu_metrics PRIVATE -Wall -Wextra -Werror)
if(NOT GPU_CORE_ONLY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 egl glesv2 gbm)
    add_library(cgltf STATIC src/cgltf_impl.c)
    add_executable(gpu-benchmark src/main.cpp src/renderer.cpp src/model.cpp)
    target_link_libraries(gpu-benchmark PRIVATE gpu_metrics cgltf PkgConfig::APP)
    target_compile_options(gpu-benchmark PRIVATE -Wall -Wextra -Werror)
    install(TARGETS gpu-benchmark RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES data/gpu-benchmark.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES data/gpu-benchmark.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    install(FILES data/gpu-benchmark.png DESTINATION ${CMAKE_INSTALL_DATADIR}/weston)
    install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/gpu-benchmark)
    install(PROGRAMS scripts/run-acceptance DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME gpu-benchmark-acceptance)
    install(DIRECTORY data/models DESTINATION ${CMAKE_INSTALL_DATADIR}/gpu-benchmark)
    install(FILES src/third_party/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/gpu-benchmark RENAME cgltf-LICENSE)
endif()
if(BUILD_TESTING)
    add_executable(gpu-core-test tests/core_test.cpp)
    target_link_libraries(gpu-core-test PRIVATE gpu_metrics)
    target_compile_options(gpu-core-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME core COMMAND gpu-core-test)
    if(NOT GPU_CORE_ONLY)
        add_test(NAME self-test COMMAND gpu-benchmark --self-test)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:gpu-benchmark>)
    endif()
endif()
