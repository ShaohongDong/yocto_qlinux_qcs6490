cmake_minimum_required(VERSION 3.22)
project(hevc_benchmark LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(HEVC_CORE_ONLY "Build hardware-independent tests only" OFF)
add_library(hevc_metrics src/metrics.cpp)
target_include_directories(hevc_metrics PUBLIC src)
target_compile_features(hevc_metrics PUBLIC cxx_std_17)
target_compile_options(hevc_metrics PRIVATE -Wall -Wextra -Werror)
if(NOT HEVC_CORE_ONLY)
    find_package(PkgConfig REQUIRED)
    find_package(Threads REQUIRED)
    pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0)
    add_executable(hevc-benchmark src/main.cpp src/engine.cpp src/ui.cpp)
    target_link_libraries(hevc-benchmark PRIVATE hevc_metrics PkgConfig::APP Threads::Threads)
    target_compile_options(hevc-benchmark PRIVATE -Wall -Wextra -Werror)
    install(TARGETS hevc-benchmark RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES data/hevc-benchmark.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES data/hevc-benchmark.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    install(FILES data/hevc-benchmark.png DESTINATION ${CMAKE_INSTALL_DATADIR}/weston)
    install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/hevc-benchmark)
    install(PROGRAMS scripts/run-latency-matrix DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME hevc-benchmark-latency-matrix)
    install(PROGRAMS scripts/run-latency DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME hevc-benchmark-latency)
    install(PROGRAMS scripts/run-acceptance DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME hevc-benchmark-acceptance)
endif()
if(BUILD_TESTING)
    add_executable(hevc-metrics-test tests/metrics_test.cpp)
    target_link_libraries(hevc-metrics-test PRIVATE hevc_metrics)
    target_compile_options(hevc-metrics-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME metrics COMMAND hevc-metrics-test)
    if(NOT HEVC_CORE_ONLY)
        add_executable(hevc-engine-test tests/engine_test.cpp)
        target_link_libraries(hevc-engine-test PRIVATE hevc_metrics PkgConfig::APP Threads::Threads)
        target_compile_options(hevc-engine-test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME engine-accounting COMMAND hevc-engine-test)
        add_test(NAME no-hardware-self-test COMMAND hevc-benchmark --self-test)
    endif()
endif()

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME latency-runner COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/latency_runner_test.py)
    if(NOT HEVC_CORE_ONLY)
        add_test(NAME latency-invalid-profile COMMAND hevc-benchmark --latency-profile invalid)
        set_tests_properties(latency-invalid-profile PROPERTIES WILL_FAIL TRUE)
        add_test(NAME latency-invalid-rate COMMAND hevc-benchmark --operating-rate 31)
        set_tests_properties(latency-invalid-rate PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
