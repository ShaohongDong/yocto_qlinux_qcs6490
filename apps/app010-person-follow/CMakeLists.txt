cmake_minimum_required(VERSION 3.22)
project(person_follow LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-pbutils-1.0)
find_path(QNN_INCLUDE_DIR QnnInterface.h PATH_SUFFIXES QNN REQUIRED)
find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3 REQUIRED)
add_library(bytetrack STATIC
    third_party/bytetrack/src/BYTETracker.cpp third_party/bytetrack/src/STrack.cpp
    third_party/bytetrack/src/KalmanFilter.cpp third_party/bytetrack/src/lapjv.cpp
    third_party/bytetrack/src/Object.cpp third_party/bytetrack/src/Rect.cpp)
target_compile_features(bytetrack PUBLIC cxx_std_17)
target_include_directories(bytetrack PUBLIC third_party/bytetrack/include)
target_include_directories(bytetrack SYSTEM PUBLIC ${EIGEN3_INCLUDE_DIR})
add_executable(person-follow src/main.cpp src/qnn.cpp)
target_compile_features(person-follow PRIVATE cxx_std_17)
target_compile_options(person-follow PRIVATE -Wall -Wextra -Werror)
target_include_directories(person-follow PRIVATE ${QNN_INCLUDE_DIR})
target_link_libraries(person-follow PRIVATE bytetrack PkgConfig::APP Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS person-follow RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/person-follow.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/person-follow.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/person-follow)
install(FILES third_party/bytetrack/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/person-follow RENAME LICENSE-ByteTrack)
if(BUILD_TESTING)
    add_executable(follow-replay tests/follow_replay.cpp)
    target_compile_features(follow-replay PRIVATE cxx_std_17)
    target_include_directories(follow-replay PRIVATE src)
    add_executable(follow-test tests/follow_test.cpp)
    target_compile_features(follow-test PRIVATE cxx_std_17)
    target_include_directories(follow-test PRIVATE src)
    add_test(NAME follow COMMAND follow-test)
    add_executable(tracker-replay tests/tracker_replay.cpp)
    target_compile_features(tracker-replay PRIVATE cxx_std_17)
    target_compile_options(tracker-replay PRIVATE -Wall -Wextra -Werror)
    target_include_directories(tracker-replay PRIVATE src)
    target_link_libraries(tracker-replay PRIVATE bytetrack)
    add_executable(pose-contract-probe tests/contract_probe.cpp)
    target_compile_features(pose-contract-probe PRIVATE cxx_std_17)
    target_include_directories(pose-contract-probe PRIVATE src)
    add_executable(ai-core-test tests/core_test.cpp)
    target_compile_features(ai-core-test PRIVATE cxx_std_17)
    target_compile_options(ai-core-test PRIVATE -Wall -Wextra -Werror)
    target_include_directories(ai-core-test PRIVATE src)
    target_link_libraries(ai-core-test PRIVATE bytetrack)
    add_test(NAME core COMMAND ai-core-test)
    add_test(NAME self-test COMMAND person-follow --self-test)
    add_test(NAME invalid-confidence COMMAND person-follow --confidence nan)
    set_tests_properties(invalid-confidence PROPERTIES WILL_FAIL TRUE)
endif()
