cmake_minimum_required(VERSION 3.22)
project(wifi_test LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 libnm>=1.20)
add_library(wifi-core src/core.cpp)
target_include_directories(wifi-core PUBLIC src)
target_compile_features(wifi-core PUBLIC cxx_std_17)
target_compile_options(wifi-core PRIVATE -Wall -Wextra -Werror)
target_link_libraries(wifi-core PUBLIC PkgConfig::APP)
add_executable(wifi-test src/backend.cpp src/main.cpp src/ui.cpp)
target_compile_options(wifi-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(wifi-test PRIVATE wifi-core Threads::Threads)
install(TARGETS wifi-test RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/wifi-test.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/wifi-test.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES data/wifi-test.png DESTINATION ${CMAKE_INSTALL_DATADIR}/weston)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/wifi-test)
if(BUILD_TESTING)
    add_test(NAME self-test COMMAND wifi-test --self-test)
    add_executable(wifi-core-test tests/core_test.cpp)
    target_link_libraries(wifi-core-test PRIVATE wifi-core)
    target_compile_options(wifi-core-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME core COMMAND wifi-core-test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:wifi-test>)
endif()
