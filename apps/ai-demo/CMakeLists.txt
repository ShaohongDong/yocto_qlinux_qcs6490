cmake_minimum_required(VERSION 3.22)
project(ai_demo LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-pbutils-1.0)
find_path(QNN_INCLUDE_DIR QnnInterface.h PATH_SUFFIXES QNN REQUIRED)
add_executable(ai-demo src/main.cpp src/qnn.cpp)
target_compile_features(ai-demo PRIVATE cxx_std_17)
target_compile_options(ai-demo PRIVATE -Wall -Wextra -Werror)
target_include_directories(ai-demo PRIVATE ${QNN_INCLUDE_DIR})
target_link_libraries(ai-demo PRIVATE PkgConfig::APP Threads::Threads ${CMAKE_DL_LIBS})
install(TARGETS ai-demo RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/ai-demo.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/ai-demo.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/ai-demo)
if(BUILD_TESTING)
    add_executable(ai-core-test tests/core_test.cpp)
    target_compile_features(ai-core-test PRIVATE cxx_std_17)
    target_compile_options(ai-core-test PRIVATE -Wall -Wextra -Werror)
    target_include_directories(ai-core-test PRIVATE src)
    add_test(NAME core COMMAND ai-core-test)
    add_test(NAME self-test COMMAND ai-demo --self-test)
    add_test(NAME invalid-confidence COMMAND ai-demo --confidence nan)
    set_tests_properties(invalid-confidence PROPERTIES WILL_FAIL TRUE)
endif()
