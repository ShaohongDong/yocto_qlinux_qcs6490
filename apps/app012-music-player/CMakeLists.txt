cmake_minimum_required(VERSION 3.22)
project(music_player LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
# The selected-app recipe does not inherit pkgconfig; isolate target lookup.
if(CMAKE_CROSSCOMPILING AND CMAKE_SYSROOT)
    set(ENV{PKG_CONFIG_SYSROOT_DIR} "${CMAKE_SYSROOT}")
    set(ENV{PKG_CONFIG_LIBDIR} "${CMAKE_SYSROOT}/usr/lib/pkgconfig:${CMAKE_SYSROOT}/usr/share/pkgconfig")
    set(ENV{PKG_CONFIG_PATH} "")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(APP REQUIRED IMPORTED_TARGET gtk+-3.0 gstreamer-1.0 libpulse)
add_executable(music-player src/main.cpp)
target_compile_features(music-player PRIVATE cxx_std_17)
target_compile_options(music-player PRIVATE -Wall -Wextra -Werror)
target_link_libraries(music-player PRIVATE PkgConfig::APP)
install(TARGETS music-player RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/music-player.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/music-player.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/music-player)
if(BUILD_TESTING)
    add_test(NAME self-test COMMAND music-player --self-test)
    add_test(NAME unknown-option COMMAND music-player --unknown)
    set_tests_properties(unknown-option PROPERTIES WILL_FAIL TRUE)
    add_executable(playback-test tests/playback.cpp)
    target_compile_features(playback-test PRIVATE cxx_std_17)
    target_compile_options(playback-test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(playback-test PRIVATE PkgConfig::APP)
    add_test(NAME playback COMMAND playback-test)
    add_executable(output-test tests/output.cpp)
    target_compile_features(output-test PRIVATE cxx_std_17)
    target_compile_options(output-test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(output-test PRIVATE PkgConfig::APP)
    add_test(NAME output-routing COMMAND output-test)
    if(NOT CMAKE_CROSSCOMPILING)
        find_package(Python3 COMPONENTS Interpreter)
        find_program(PIPEWIRE_EXECUTABLE pipewire)
        find_program(PIPEWIRE_PULSE_EXECUTABLE pipewire-pulse)
        if(Python3_FOUND AND PIPEWIRE_EXECUTABLE AND PIPEWIRE_PULSE_EXECUTABLE)
            add_test(NAME output-reconnect COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pulse_reconnect.py
                $<TARGET_FILE:output-test> ${PIPEWIRE_EXECUTABLE} ${PIPEWIRE_PULSE_EXECUTABLE})
            set_tests_properties(output-reconnect PROPERTIES TIMEOUT 35)
        endif()
    endif()
endif()
